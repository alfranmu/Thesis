\section{Representation variety}
For this chapter we follow \cite{maret_note_nodate, casimiroCharacterVarietiesTheir2026}. Throughout, $\Gamma$ denotes a finitely generated group and $G$ a (real or complex) Lie group.

\begin{definition}
    The \emph{representation variety} of $\Gamma$ in $G$ is the set
    \[
        \Hom(\Gamma, G) = \{ \rho : \Gamma \to G \mid \rho \text{ is a group homomorphism} \}.
    \]
    Its elements are called \emph{representations}.
\end{definition}

We endow $\Hom(\Gamma, G)$ with the subspace topology induced from the compact-open topology on the space of continuous maps $\Gamma \to G$, where $\Gamma$ carries the discrete topology. Since the compact subsets of a discrete space are the finite ones, a subbasis for this topology is given by the sets
\[
    V(K, U) = \{ \rho \in \Hom(\Gamma, G) : \rho(K) \subseteq U \},
\]
where $K \subseteq \Gamma$ is finite and $U \subseteq G$ is open.

Fix a set of generators $\{\gamma_1, \dots, \gamma_n\}$ of $\Gamma$. It is customary to identify the representation variety with the space of tuples of images of the generators,
\[
    X(\Gamma, G) = \{ (\rho(\gamma_1), \dots, \rho(\gamma_n)) : \rho \in \Hom(\Gamma, G) \} \subseteq G^n.
\]
The following proposition justifies both this identification and the name ``variety''.


\begin{proposition} \label{prop:analytic-structure}
 If $G$ is a Lie group endowed with an analityc atlas then $X(\Gamma,G)$ is an analytic subvariety of $G^n$, homeomorphic to $\Hom(\Gamma,G)$. In particular $\Hom(\Gamma,G)$ inherits a natural structure of analityc variety wich is independent of the choice of generators of $\Gamma$.  	
\end{proposition}

\begin{proof}
	Let $\{\gamma_{1}, \dots, \gamma_n\}$ be a set of generators of $\gamma$, and let $r_i$ denote the set of relations for these generators. Recall that every group homomorphis is defined by the image of its generators, and those images have to preserve the relations between the generators. Each relation can be thought as a map $r_i:\Gamma^n \to \Gamma$, since this map is just a composition of inversions and multiplications, then each $r_i$ is analytic. Then $X(\Gamma,G)$ is precisely the subset of $(g_{1},\dots,g_n) \in G^n$ defined by the conditions $r_i(g_{1},\dots,g_n)=1$. It is clear that we have an isomorphism
	\[
		\Psi: \Hom(\Gamma,G) \to X(\Gamma,G), \quad \rho \to (\rho(\gamma_{1}),\dots,\rho(\gamma_n))
	\]
To see that $\Psi$ is a homeomorphism, consider $U_{1},\dots,U_n$ open sets in $G$. Then 
\[
	\Psi^{-1}(X(\Gamma,G)\cap \left(U_1 \times \dots \times U_n\right)) = \Hom(\Gamma,G) \cap \bigcap_{i=1}^n V \left(\{\gamma_i)\}, U_i \right)
\]
This proves $\Psi$ is continuous. Now for its inverse, take $K \subset \Gamma$ finite and $U \subset G$ open. Any $\gamma \in K$ can be expressed as a word in the generators, and hence it determines an analytic map $\gamma:G^n \to G $, hence
\[
	\Psi(\Hom(\Gamma,G)\cap V(K,U)) = X(\Gamma,G) \cap \bigcap_{\gamma \in K} \gamma ^{-1}(U)
\]
Therefore, $\Psi$ is a homeomorphism.

If $(\gamma'_{1},\dots,\gamma'_n)$ is another set of generators of $\Gamma$ and $X'(\Gamma,G)$ is the associated space, then the map from $X(\Gamma,G)$ to $X(\Gamma,G)$ defined as the composition 
\[
	X(\Gamma,G) \to \Hom(\Gamma,G) \to X'(\Gamma,G)
\]is an isomorphism of analytic varieties. Indeed, the map sends $(\rho(\gamma_{1}),\dots,\rho(\gamma_n))$ to $(\rho(\gamma_{1}'),\dots,\rho(\gamma_n))$ . Now, since $\gamma_i'$ is a word in the generators $\gamma_{1},\dots,\gamma_n$, it follows that $\rho(\gamma_i')$ is a word in $\rho(\gamma_{1}),\dots,\rho(\gamma_n)$. This shows that the map is analytic because word maps are analytic by assumption on $G$.

\end{proof}
\begin{proposition}
	Assume that $G$ has the structure of a real or complex algebraic group, then $X(\Gamma,G)$ is an algebraic subset of $G^n$. In particular,$\Hom(\Gamma,G) $has a natural structure of real or complex algebraic variety and the structure does not depend on the choice of generators of $\Gamma$.
\end{proposition}

\begin{proof}
    The proof is analogous to that of Proposition~\ref{prop:analytic-structure}, replacing ``analytic'' by ``regular'': word maps are regular because multiplication and inversion in $G$ are.
\end{proof}




The representation variety $\Hom(\Gamma,G)$, carries two natural symmetries given by the right action of the group $\operatorname{Aut}(\Gamma)$ of automorphisms of $\Gamma$ by pre-composition and the left action of $\operatorname{Aut}(G)$ by post-composition. Both actions preserve the analytic or algebraic structure of $\Hom(\Gamma,G)$ as a consequence of the next proposition
\begin{proposition}
	Let $\Gamma, \Gamma_{1},\Gamma_{2}$ be finitely generated groups, and let $G,G_{1},G_{2}$ be Lie or algebraic groups.
	\begin{enumerate}
	   \item If $f:\Gamma_{1} \to \Gamma_{2}$ is a homomorphism, then the induced map $ f^* : \Hom(\Gamma_{2}, G) \to \Hom(\Gamma_{1}, G),\quad \rho \mapsto \rho \circ f$  is analytic (or regular). 
	   \item If $\phi:G_{1}\to G_{2}$ is a homomorphism, then the induced map $ \phi^* : \Hom(\Gamma, G_{1}) \to \Hom(\Gamma, G_{2}), \rho \mapsto \phi \circ \rho$ is analytic (or regular).		   
	\end{enumerate}
\end{proposition}
\begin{proof}
	See \cite[Lemma 1.2.7]{maret_note_nodate}
\end{proof} 

Among $\operatorname{Aut(G)}$ normal subgroups, one is particularly important, the subgroup of inner automorphisms $\operatorname{Inn}(G)$ defined by conjugation with elements of $G$. In  particular, $\Inn(G)=G/Z(G)$. The action of $\operatorname{Inn}(G)$ on $\Hom(\Gamma, G)$ is fundamental in many geometric situations: for example, holonomy representations of surface structures are only well-defined up to conjugation by G. \subsection{Characterizations of the orbits of the $\operatorname{Inn}(G)$-action }

For a representation $\rho \in \Hom(\Gamma, G)$ we write $Z(\rho) := Z(\rho(\Gamma))$ for the centralizer of its image. Note that $Z(G) \subseteq Z(\rho)$ always, and that $Z(c_g \cdot \rho) = g Z(\rho) g^{-1}$.

\begin{definition}
    A representation $\rho \in \Hom(\Gamma, G)$ is called \emph{regular} if $\dim Z(\rho) = \dim Z(G)$, and \emph{very regular} if $Z(\rho) = Z(G)$. We denote by $\Homreg(\Gamma, G)$ and $\Homvreg(\Gamma, G)$ the subsets of regular and very regular representations, respectively.
\end{definition}

Both subsets are $\Inn(G)$-invariant: conjugation preserves dimensions, and $g Z(G) g^{-1} = Z(G)$ for every $g \in G$.
 
\begin{lemma}\label{lem:free-action}
    The action of $\Inn(G)$ on $\Homvreg(\Gamma, G)$ is free.
\end{lemma}

\begin{proof}
    Let $\rho \in \Homvreg(\Gamma, G)$ and suppose that $c_g \cdot \rho = \rho$, that is, $g \rho(\gamma) g^{-1} = \rho(\gamma)$ for all $\gamma \in \Gamma$. Then $g \in Z(\rho) = Z(G)$, so $c_g(h) = ghg^{-1} = h$ for every $h \in G$. Hence $c_g$ is the identity of $\Inn(G)$.
\end{proof}


\begin{proposition}[\cite{goldmanSymplecticNatureFundamental1984}]
	The action of $\Inn(G)$ on $\Hom(\Gamma,G)$ is locally free at $\phi$ if and only  if  $\operatorname{dim} Z(G)= \dim Z(\phi)$.
\end{proposition}
\begin{proof}
  See	\cite[Proposition 8]{casimiroCharacterVarietiesTheir2026}.
\end{proof}

To obtain a well-behaved quotient we also need the action to be proper.
\begin{definition}
	A proper group action is a continuous group action of a topological group $G$ on topological space $X$ for wich the map \[
		G \times X \to X\times X, (g,x) \to (gx,x)
	\]
	is a proper map, i.e., inverse images of compact sets are compact.
\end{definition}


\begin{definition}
	let $G$ be an algebraic group. A Borel subgroup of $G$ is a maximal, Zariski closed, solvable connected subgroup of $G$. A Zariski closed subgroup that contains a Borel subgroup is called a parabolic subgroup.
\end{definition}


\begin{definition}
	Let $G$ be an algebraic group. A representation $\phi: \Gamma \to G$ is called irreducible if the image of $\phi$ does not lie in a proper parabolic subgroup of $G$. We denote by $\Hom ^{\text{irr}}(\Gamma,G)$ the $\Inn(G)-$invariant subspace of irreducible representations.
\end{definition}


\begin{theorem}\label{thm:proper-action}
	Let $G$ be a reductive algebraic group. The $\Inn(G)$-action on $\Hom ^{\text{irr}}(\Gamma,G)$  is proper.
\end{theorem}

\begin{proof}
    See \cite[Prop.~1.1]{JohnsonDennis1987DSAt}.
\end{proof}


Properness guarantees that the quotient by $\Inn(G)$ behaves well topologically: the orbits are closed and the quotient space is Hausdorff.

\begin{definition}
    Let $G$ be an algebraic group. A representation $\rho : \Gamma \to G$ is called \emph{good} if it is irreducible and very regular. We denote by $\Homgd(\Gamma, G)$ the $\Inn(G)$-invariant subset of good representations.
\end{definition}


By Lemma~\ref{lem:free-action} and Theorem~\ref{thm:proper-action}, the conjugation action on $\Homgd(\Gamma, G)$ is both free and proper. Hence, whenever $\Homgd(\Gamma, G)$ is a manifold, the quotient by conjugation inherits the structure of an analytic manifold. This happens, for instance, for closed surface groups.

\begin{proposition}\label{prop:good-surface}
    Let $G$ be a reductive algebraic group.
    \begin{enumerate}
        \item $\Homgd(\Gamma, G)$ is Zariski open in $\Hom(\Gamma, G)$.
        \item If $\Gamma = \pi_{g,0}$, then:
        \begin{enumerate}
            \item $\Homgd(\pi_{g,0}, G)$ is an analytic manifold of dimension $(2g-1)\dim G + \dim Z(G)$;
            \item the conjugation action on $\Homgd(\pi_{g,0}, G)$ is free and proper;
            \item the quotient $\Homgd(\pi_{g,0}, G) / \Inn(G)$ is an analytic manifold of dimension $(2g-2)\dim G + 2\dim Z(G)$, which is always even.
        \end{enumerate}
    \end{enumerate}
\end{proposition}

\begin{definition}
    An algebraic group is called \emph{linearly reductive} if all its finite-dimensional representations are completely reducible, i.e.\ decompose as direct sums of simple subrepresentations. A subgroup of an algebraic group is called \emph{completely reducible} if its Zariski closure is linearly reductive.
\end{definition}

Over $\RR$ or $\CC$, an algebraic group is linearly reductive if and only if its Zariski identity component is reductive (see \cite[]). This equivalence allows us to use both notions interchangeably in most geometric contexts.

\begin{definition}
    Let $G$ be an algebraic group. A representation $\rho : \Gamma \to G$ is called \emph{reductive} (or \emph{completely reducible}) if $\rho(\Gamma) \subseteq G$ is completely reducible. We denote by $\Homred(\Gamma, G)$ the $\Inn(G)$-invariant subset of reductive representations.
\end{definition}

\begin{proposition}[\cite{TODO-ref-17, TODO-ref-47}]\label{prop:closed-orbits}
    Let $G$ be a reductive algebraic group. A representation $\rho : \Gamma \to G$ is reductive if and only if its conjugation orbit $\mathcal{O}_\rho$ is closed in $\Hom(\Gamma, G)$.
\end{proposition}
 
As a consequence of Proposition~\ref{prop:closed-orbits}, the topological quotient $\Homred(\Gamma, G) / \Inn(G)$ is a $T_1$-space. Moreover, it is in fact Hausdorff (see \cite{TODO-ref-45}). This provides a natural and geometrically meaningful setting in which to define the character variety associated with the pair $(\Gamma, G)$.
